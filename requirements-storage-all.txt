# The window the ACME stack works in: certbot and acme 5.8.0 require
# cryptography>=47 and pyopenssl 26.4.0 requires >=49,<51. This file is
# documented as installable on its own, and standalone nothing else constrains
# cryptography: with a bare floor it would resolve to the newest release, which
# pyopenssl then refuses. The Azure Key Vault Certificate-object path needs only
# `load_pem_x509_certificates` (plural, added in 36.0) and the PKCS12 builder,
# both inside that window. The authoritative reason lives beside the pin in
# requirements.txt; move this window when that moves.
cryptography>=49,<51

# AWS Secrets Manager
boto3>=1.43.108

# HashiCorp Vault
hvac>=2.4.0

# Infisical
# Pinned, and not to the latest. infisical-python 2.3.6 ships no
# manylinux x86_64 wheel and no sdist — only aarch64, macOS and Windows —
# so `>=2.3.6` cannot be installed on the amd64 image at all, while the
# arm64 image installs it fine. 2.3.5 is the last release with complete
# platform coverage. This is the only compiled dependency in the storage
# extras; the rest ship pure-Python wheels and are architecture-neutral.
infisical-python==2.3.5
