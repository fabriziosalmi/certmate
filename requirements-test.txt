# The test toolchain, pinned. These floated unpinned, so every CI run resolved
# whatever was newest that day and two runs a few hours apart did not install
# the same tools — which is how a flaky native crash in the coverage core could
# appear out of nowhere against an otherwise identical dependency set. Pinned to
# the set the suite was last green on so a red is the code's fault, not the day's
# releases; Dependabot proposes the bumps, and a bump that breaks the suite is
# then a reviewable PR rather than a silent regression. (These are not hash-locked
# like requirements*.lock because of the editable installs below.)
pytest==9.1.1
pytest-cov==7.1.0
pytest-mock==3.17.0
pytest-asyncio==1.4.0
pytest-flask==1.3.0
pytest-xdist==3.8.0
pytest-playwright==0.10.0
coverage==7.16.2
requests-mock==1.12.1
freezegun==1.5.5
playwright==1.64.0
# In-repo terminal clients (SDK + CLI), installed editable so the e2e gate can
# exercise them against the test container and keep them in lockstep with the
# server they wrap. Order matters: the SDK first (the CLI depends on it).
-e ./clients/certmate-sdk
-e ./clients/certmate-cli

# Read by tests/test_dependabot_holds_the_stack.py to parse .github/dependabot.yml.
# Pinned explicitly rather than relied on transitively: it arrives today via
# bandit, cloudflare and dns-lexicon, so the test would start failing with
# ModuleNotFoundError the day any of those drops it — for reasons having
# nothing to do with what it checks.
PyYAML==6.0.3

# Read by tests/test_no_requirements_file_can_break_the_acme_stack.py to compare
# version specifiers. Same reasoning as PyYAML above: it arrives transitively
# today (pip, build tooling), so relying on that would break the test the day
# something drops it, for reasons unrelated to what it checks.
packaging==26.3

# Read by tests/test_complexity_ratchet_is_tight.py, which shells out to flake8
# to measure the tree's worst cyclomatic complexity and compares it with the
# number CI enforces. Pinned to the version the lint step installs so the test
# job and the lint job cannot disagree about what "too complex" means.
flake8==7.4.1
# Read by scripts/check_ruff_budget.py, the ruff ratchet CI and `make lint` run (#1093). Pinned
# exactly, and to the version the baseline in that script was measured with: rules change between
# releases, so a bump (Dependabot proposes one) fails that check until the baseline is re-measured
# (`python scripts/check_ruff_budget.py --print`) and RUFF_VERSION moves with this pin.
ruff==0.17.0
# Read by scripts/check_mypy_budget.py, the mypy ratchet (#1094). Pinned exactly, to the version its
# baseline was measured with: what mypy reports changes between releases, so a bump (Dependabot
# proposes one) fails that check until the baseline is re-measured (`python scripts/check_mypy_budget.py
# --print`) and MYPY_VERSION moves with this pin.
mypy==2.4.0
